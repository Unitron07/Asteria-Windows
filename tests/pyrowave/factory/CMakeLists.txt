cmake_minimum_required(VERSION 3.27)
project(pyrowave_factory_ownership LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
# Compile the actual patched pinned function bodies, never a handwritten copy.
file(READ "${PYROWAVE_SOURCE}/pyrowave_c.cpp" source)
string(FIND "${source}" "pyrowave_result pyrowave_create_device(const" begin)
string(FIND "${source}" "void pyrowave_device_report_performance_stats(" end)
if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Pinned borrowed factory boundaries changed")
endif()
math(EXPR length "${end}-${begin}")
string(SUBSTRING "${source}" ${begin} ${length} factory)
string(FIND "${source}" "void pyrowave_device_destroy(" begin)
string(FIND "${source}" "struct pyrowave_sync_object_opaque" end)
math(EXPR length "${end}-${begin}")
string(SUBSTRING "${source}" ${begin} ${length} destroy)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pinned_factory.inc" "${factory}\n${destroy}")
file(SHA256 "${CMAKE_CURRENT_BINARY_DIR}/pinned_factory.inc" extracted_sha)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/extracted-source.sha256" "${extracted_sha}\n")
add_executable(pyrowave-factory-cleanup-tests factory_ownership.cpp)
target_include_directories(pyrowave-factory-cleanup-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PYROWAVE_SOURCE}" "${VULKAN_HEADERS}")
add_test(NAME pinned-factory-ownership COMMAND pyrowave-factory-cleanup-tests)
