cmake_minimum_required(VERSION 3.27)
project(asteria_isolated_vulkan_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
set(VIDEO "${CMAKE_CURRENT_SOURCE_DIR}/../../../app/streaming/video")
add_executable(pyrowave-vulkan-policy-tests policy_tests.cpp)
target_include_directories(pyrowave-vulkan-policy-tests PRIVATE ${VIDEO})
add_test(NAME vulkan-policy COMMAND pyrowave-vulkan-policy-tests)
add_executable(pyrowave-vulkan-shared-device-policy-tests shared_policy_tests.cpp)
add_test(NAME vulkan-shared-policy COMMAND pyrowave-vulkan-shared-device-policy-tests)
option(PYROWAVE_VULKAN_SHARED_DEVICE "Build offline same-device pinned codec proof" OFF)
if(PYROWAVE_VULKAN_SHARED_DEVICE)
    add_executable(stage3-owner-runner-fixture owner_runner_fixture.cpp)
    if(NOT WIN32 OR NOT SDL_ARCH MATCHES "^(x64|arm64)$" OR
       NOT EXISTS "${PYROWAVE_ROOT}/include/pyrowave/pyrowave.h" OR
       NOT EXISTS "${VULKAN_HEADERS}/vulkan/vulkan_core.h")
        message(FATAL_ERROR "Stage 3 requires Windows, SDL_ARCH, pinned PYROWAVE_ROOT and VULKAN_HEADERS")
    endif()
    foreach(input IN ITEMS include/${SDL_ARCH}/SDL2/SDL.h lib/${SDL_ARCH}/SDL2.lib lib/${SDL_ARCH}/SDL2.dll lib/${SDL_ARCH}/SDL3.dll)
        if(NOT EXISTS "${SDL_ROOT}/${input}")
            message(FATAL_ERROR "Missing pinned SDL dependency ${input}")
        endif()
    endforeach()
    add_library(stage3-runtime STATIC ${VIDEO}/pyrowave_runtime.cpp ${VIDEO}/pyrowave_frame.cpp)
    target_include_directories(stage3-runtime PUBLIC ${VIDEO} ${VULKAN_HEADERS} ${PYROWAVE_ROOT}/include)
    target_compile_definitions(stage3-runtime PUBLIC PYROWAVE_EXPERIMENTAL PYROWAVE_VULKAN_SHARED_DEVICE VK_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX)
    add_executable(pyrowave-vulkan-shared-device shared_probe.cpp shared_device.cpp ${VIDEO}/pyrowave_vulkan_dispatch.cpp ${VIDEO}/pyrowave_vulkan_probe.cpp)
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE stage3_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    target_compile_definitions(pyrowave-vulkan-shared-device PRIVATE STAGE3_SOURCE_REVISION="${stage3_revision}")
    target_include_directories(pyrowave-vulkan-shared-device PRIVATE ${SDL_ROOT}/include/${SDL_ARCH}/SDL2)
    target_link_libraries(pyrowave-vulkan-shared-device PRIVATE stage3-runtime ${SDL_ROOT}/lib/${SDL_ARCH}/SDL2.lib bcrypt psapi)
    foreach(dll IN ITEMS SDL2.dll SDL3.dll)
        add_custom_command(TARGET pyrowave-vulkan-shared-device POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${SDL_ROOT}/lib/${SDL_ARCH}/${dll} $<TARGET_FILE_DIR:pyrowave-vulkan-shared-device>)
    endforeach()
    add_library(stage3-mock SHARED shared_mock.cpp)
    target_include_directories(stage3-mock PRIVATE ${VULKAN_HEADERS} ${PYROWAVE_ROOT}/include)
    set_target_properties(stage3-mock PROPERTIES OUTPUT_NAME libpyrowave-shared-0 RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/stage3-mock)
    add_library(stage3-no-borrow SHARED shared_mock.cpp)
    target_include_directories(stage3-no-borrow PRIVATE ${VULKAN_HEADERS} ${PYROWAVE_ROOT}/include)
    target_compile_definitions(stage3-no-borrow PRIVATE STAGE3_NO_BORROW)
    set_target_properties(stage3-no-borrow PROPERTIES OUTPUT_NAME libpyrowave-shared-0 RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/stage3-no-borrow)
    add_executable(pyrowave-vulkan-shared-runtime-tests shared_runtime_tests.cpp)
    target_link_libraries(pyrowave-vulkan-shared-runtime-tests PRIVATE stage3-runtime)
    add_test(NAME vulkan-shared-runtime-mocks COMMAND pyrowave-vulkan-shared-runtime-tests $<TARGET_FILE_DIR:stage3-mock> $<TARGET_FILE_DIR:stage3-no-borrow>)
    add_test(NAME vulkan-shared-real-gpu COMMAND pyrowave-vulkan-shared-device --runtime ${PYROWAVE_ROOT}/bin --log stage3-gpu.log)
    set_tests_properties(vulkan-shared-real-gpu PROPERTIES SKIP_RETURN_CODE 77 LABELS "gpu;opt-in" TIMEOUT 180)
endif()
option(PYROWAVE_VULKAN_PROBE "Build isolated synthetic Windows GPU probe (no codec)" OFF)
if(PYROWAVE_VULKAN_PROBE)
    if(NOT WIN32 OR NOT SDL_ARCH MATCHES "^(x64|arm64)$")
        message(FATAL_ERROR "Probe requires Windows and SDL_ARCH=x64 or arm64")
    endif()
    if(NOT EXISTS "${VULKAN_HEADERS}/vulkan/vulkan_core.h")
        message(FATAL_ERROR "Set VULKAN_HEADERS to the reviewed pinned include directory")
    endif()
    foreach(input IN ITEMS include/${SDL_ARCH}/SDL2/SDL.h lib/${SDL_ARCH}/SDL2.lib lib/${SDL_ARCH}/SDL2.dll lib/${SDL_ARCH}/SDL3.dll)
        if(NOT EXISTS "${SDL_ROOT}/${input}")
            message(FATAL_ERROR "Missing pinned SDL dependency ${SDL_ROOT}/${input}")
        endif()
    endforeach()
    add_executable(pyrowave-vulkan-probe main.cpp ${VIDEO}/pyrowave_vulkan_dispatch.cpp ${VIDEO}/pyrowave_vulkan_probe.cpp)
    target_include_directories(pyrowave-vulkan-probe PRIVATE ${VIDEO} ${VULKAN_HEADERS} ${SDL_ROOT}/include/${SDL_ARCH}/SDL2)
    target_compile_definitions(pyrowave-vulkan-probe PRIVATE VK_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(pyrowave-vulkan-probe PRIVATE ${SDL_ROOT}/lib/${SDL_ARCH}/SDL2.lib)
    # No Vulkan or PyroWave import library, no production app target changes.
    foreach(dll IN ITEMS SDL2.dll SDL3.dll)
        add_custom_command(TARGET pyrowave-vulkan-probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${SDL_ROOT}/lib/${SDL_ARCH}/${dll} $<TARGET_FILE_DIR:pyrowave-vulkan-probe>)
    endforeach()
    add_test(NAME vulkan-hidden-smoke COMMAND pyrowave-vulkan-probe --hidden --seconds 2 --log hidden-smoke.log)
    set_tests_properties(vulkan-hidden-smoke PROPERTIES SKIP_RETURN_CODE 77 LABELS "gpu;opt-in" TIMEOUT 30)
endif()
