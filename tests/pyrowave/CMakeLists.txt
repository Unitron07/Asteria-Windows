cmake_minimum_required(VERSION 3.27)
project(asteria_pyrowave_offline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(VIDEO ${CMAKE_CURRENT_SOURCE_DIR}/../../app/streaming/video)
set(COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../../moonlight-common-c/moonlight-common-c/src)
add_executable(pyrowave-negotiation-tests negotiation_tests.c)
target_include_directories(pyrowave-negotiation-tests PRIVATE ${COMMON})
add_test(NAME pyrowave-negotiation COMMAND pyrowave-negotiation-tests)
add_library(pyrowave_frame STATIC ${VIDEO}/pyrowave_frame.cpp)
target_include_directories(pyrowave_frame PUBLIC ${VIDEO})
add_executable(pyrowave-parser-tests parser_tests.cpp)
target_link_libraries(pyrowave-parser-tests PRIVATE pyrowave_frame)
add_test(NAME pyrowave-parser COMMAND pyrowave-parser-tests)
add_executable(pyrowave-live-frame-tests live_frame_tests.cpp)
target_include_directories(pyrowave-live-frame-tests PRIVATE ${COMMON})
target_link_libraries(pyrowave-live-frame-tests PRIVATE pyrowave_frame)
add_test(NAME pyrowave-live-frame COMMAND pyrowave-live-frame-tests)
add_executable(pyrowave-legacy-parser-tests legacy_parser_tests.cpp)
target_link_libraries(pyrowave-legacy-parser-tests PRIVATE pyrowave_frame)
add_test(NAME pyrowave-legacy-offline-parser COMMAND pyrowave-legacy-parser-tests)
add_executable(pyrowave-presentation-tests presentation_tests.cpp)
target_include_directories(pyrowave-presentation-tests PRIVATE ${VIDEO})
add_test(NAME pyrowave-presentation-patterns COMMAND pyrowave-presentation-tests)
option(PYROWAVE_EXPERIMENTAL "Build offline Windows runtime/roundtrip probe" OFF)
option(PYROWAVE_SDL_PRESENTATION "Include offline SDL2 presentation qualification modes" OFF)
if(PYROWAVE_SDL_PRESENTATION AND NOT PYROWAVE_EXPERIMENTAL)
    message(FATAL_ERROR "SDL qualification requires the explicit offline experiment")
endif()
if(PYROWAVE_EXPERIMENTAL)
    if(NOT WIN32)
        message(FATAL_ERROR "The P0 runtime experiment is Windows only")
    endif()
    if(NOT EXISTS "${PYROWAVE_ROOT}/include/pyrowave/pyrowave.h" OR
       NOT EXISTS "${VULKAN_HEADERS}/vulkan/vulkan_core.h")
        message(FATAL_ERROR "Set PYROWAVE_ROOT to the target install and VULKAN_HEADERS to pinned include/")
    endif()
    add_library(pyrowave_runtime STATIC ${VIDEO}/pyrowave_runtime.cpp)
    target_compile_definitions(pyrowave_runtime PUBLIC PYROWAVE_EXPERIMENTAL=1)
    target_include_directories(pyrowave_runtime PUBLIC ${VIDEO} ${PYROWAVE_ROOT}/include ${VULKAN_HEADERS})
    target_link_libraries(pyrowave_runtime PUBLIC pyrowave_frame)
    add_executable(pyrowave-offline-proof offline_proof.cpp)
    target_link_libraries(pyrowave-offline-proof PRIVATE pyrowave_runtime)
    if(PYROWAVE_SDL_PRESENTATION)
        # Consume the same verified v15 target archive as the normal Windows build.
        # SDL_ROOT is libs/windows; SDL_ARCH is x64 or arm64. No find/fallback to system SDL.
        if(NOT SDL_ARCH MATCHES "^(x64|arm64)$")
            message(FATAL_ERROR "Set SDL_ARCH to the selected target architecture")
        endif()
        foreach(input IN ITEMS include/${SDL_ARCH}/SDL2/SDL.h lib/${SDL_ARCH}/SDL2.lib lib/${SDL_ARCH}/SDL2.dll lib/${SDL_ARCH}/SDL3.dll)
            if(NOT EXISTS "${SDL_ROOT}/${input}")
                message(FATAL_ERROR "Missing pinned SDL dependency: ${SDL_ROOT}/${input}")
            endif()
        endforeach()
        target_sources(pyrowave-offline-proof PRIVATE sdl_presentation.cpp)
        target_compile_definitions(pyrowave-offline-proof PRIVATE PYROWAVE_SDL_PRESENTATION=1)
        target_include_directories(pyrowave-offline-proof PRIVATE ${SDL_ROOT}/include/${SDL_ARCH}/SDL2)
        target_link_libraries(pyrowave-offline-proof PRIVATE ${SDL_ROOT}/lib/${SDL_ARCH}/SDL2.lib bcrypt)
        add_executable(pyrowave-live-sdl-tests live_sdl_tests.cpp ${VIDEO}/pyrowave_sdl.cpp)
        target_include_directories(pyrowave-live-sdl-tests PRIVATE ${VIDEO} ${SDL_ROOT}/include/${SDL_ARCH}/SDL2)
        target_link_libraries(pyrowave-live-sdl-tests PRIVATE ${SDL_ROOT}/lib/${SDL_ARCH}/SDL2.lib)
        add_test(NAME pyrowave-live-sdl-range COMMAND pyrowave-live-sdl-tests)
        foreach(dll IN ITEMS SDL2.dll SDL3.dll)
            add_custom_command(TARGET pyrowave-offline-proof POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                ${SDL_ROOT}/lib/${SDL_ARCH}/${dll} $<TARGET_FILE_DIR:pyrowave-offline-proof>)
        endforeach()
    endif()
    # Never link pyrowave-shared.lib or Vulkan. DLL absence is recoverable.
    add_test(NAME pyrowave-missing-runtime COMMAND pyrowave-offline-proof --expect-missing ${CMAKE_CURRENT_BINARY_DIR}/absent)
    add_executable(pyrowave-runtime-tests runtime_tests.cpp)
    target_link_libraries(pyrowave-runtime-tests PRIVATE pyrowave_runtime)
    foreach(kind IN ITEMS wrong-version missing-exports)
        add_library(${kind} SHARED rejected_runtime.cpp)
        set_target_properties(${kind} PROPERTIES OUTPUT_NAME libpyrowave-shared-0
            RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/${kind})
        if(kind STREQUAL "wrong-version")
            target_compile_definitions(${kind} PRIVATE TEST_API_MINOR=7)
            set(reason "expected API 0.6.0")
        else()
            target_compile_definitions(${kind} PRIVATE TEST_API_MINOR=6)
            set(reason "missing C API export")
        endif()
        add_test(NAME pyrowave-${kind} COMMAND pyrowave-runtime-tests $<TARGET_FILE_DIR:${kind}> "${reason}")
    endforeach()
    if(GRANITE_ROOT AND GRANITE_BUILD)
        file(GLOB_RECURSE math_library "${GRANITE_BUILD}/granite-math.lib")
        list(LENGTH math_library math_count)
        if(NOT math_count EQUAL 1)
            message(FATAL_ERROR "Expected one target granite-math.lib for compatibility tests")
        endif()
        add_executable(granite-compatibility-tests granite_compatibility_tests.cpp)
        target_include_directories(granite-compatibility-tests PRIVATE ${GRANITE_ROOT}/math ${GRANITE_ROOT}/util)
        target_link_libraries(granite-compatibility-tests PRIVATE ${math_library})
        add_test(NAME granite-portable-compatibility COMMAND granite-compatibility-tests)
    endif()
endif()
